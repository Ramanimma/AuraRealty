Build the Monthly Market Update newsletter template for Texas Realty Hub. This is a premium, branded email template used specifically when "Monthly Market Update" is selected in the Campaigns tab.

1. Create server/lib/templates/marketUpdateEmail.ts:

ts
export function buildMarketUpdateEmail(data: {
  month: string;           // e.g. "July 2026"
  bannerImageUrl?: string; // optional, falls back to default
  medianPrice: string;     // e.g. "$485,000"
  medianPriceTrend?: string; // e.g. "▲ 1.5%" or "▼ 2%"
  daysOnMarket: string;    // e.g. "28"
  daysOnMarketTrend?: string;
  homesSold: string;       // e.g. "342"
  homesSoldTrend?: string;
  inventory: string;       // e.g. "2.1 months"
  inventoryTrend?: string;
  neighborhoodSpotlight: string;  // HTML content
  buyerInsights: string;          // HTML content
  sellerInsights: string;         // HTML content
  tipOfTheMonth: string;          // HTML content
  marketSummary: string;          // HTML content — 2-3 sentence overview
  recipientName: string;
  senderName: string;
  unsubscribeToken: string;
  senderEmail: string;
  isPreview?: boolean;
}): string
