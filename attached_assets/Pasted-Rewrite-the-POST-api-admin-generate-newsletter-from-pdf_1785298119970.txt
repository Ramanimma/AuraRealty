Extract these exact fields:

  soldListRatio: string; soldListRatioTrend: string;
  pricePerSqFt: string; pricePerSqFtTrend: string;
  inventory: string; inventoryTrend: string;
  marketSummary: string;
  neighborhoodSpotlight: string;
  buyerInsights: string;
  sellerInsights: string;
  tipOfMonth: string;
  month: string;
}): string {

  // Helper to build a stat card
  const statCard = (label: string, value: string, trend: string) => `
    <td width="30%" style="background:#F8F9FA;border:1px solid #E5E7EB;border-radius:8px;
                            padding:16px 12px;text-align:center;vertical-align:top;">
      <p style="font-family:'Helvetica Neue',Arial,sans-serif;font-size:11px;

                             color:#6B7280;margin:0;">${trend}</p>` : ''}
    </td>`;

  // Helper to build bullet points from pipe-separated string
  const bulletPoints = (text: string) => text
    ? text.split('|').map(b => b.trim()).filter(Boolean)

    : '<li>See full report for details</li>';

      <tr>

        <td width="3%"></td>
        ${statCard('Price per Sq Ft', data.pricePerSqFt, data.pricePerSqFtTrend)}
        <td width="3%"></td>
        ${statCard('Active Inventory', data.inventory, data.inventoryTrend)}
      </tr>
    </table>

    <hr style="border:none;border-top:1px solid #E5E7EB;margin:0 0 28px;">

    </h2>
    <p style="font-family:'Helvetica Neue',Arial,sans-serif;font-size:15px;

      🏡 For Buyers
    </h2>
    <ul style="font-family:'Helvetica Neue',Arial,sans-serif;font-size:15px;

      ${bulletPoints(data.buyerInsights)}

    <hr style="border:none;border-top:1px solid #E5E7EB;margin:0 0 28px;">
