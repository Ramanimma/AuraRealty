export function buildMarketUpdateNewsletterHtml(data: MarketUpdateData): string {

  function card(label: string, value: string, trend: string): string {
    return `<td style="width:30%;padding:16px;background-color:#ffffff;border:1px solid #e5e7eb;border-radius:6px;text-align:center;vertical-align:top;">
      <p style="margin:0 0 8px 0;font-size:11px;color:#6b7280;text-transform:uppercase;letter-spacing:1px;font-family:Arial,sans-serif;">${label}</p>
      <p style="margin:0 0 4px 0;font-size:26px;font-weight:700;color:#111827;font-family:Georgia,serif;">${value || 'N/A'}</p>
      ${trend ? `<p style="margin:0;font-size:12px;color:#6b7280;font-family:Arial,sans-serif;">${trend}</p>` : '<p style="margin:0;font-size:12px;color:transparent;">-</p>'}
    </td>`;
  }

  function bullets(text: string): string {
    if (!text) return '<li style="font-family:Arial,sans-serif;font-size:15px;color:#374151;margin-bottom:8px;">See report for details</li>';
    return text.split('|').map(b => b.trim()).filter(Boolean)
      .map(b => `<li style="font-family:Arial,sans-serif;font-size:15px;color:#374151;line-height:1.6;margin-bottom:8px;">${b}</li>`)
      .join('');
  }

  function heading(emoji: string, title: string): string {
    return `<p style="margin:0 0 16px 0;font-size:20px;font-weight:700;color:#111827;font-family:Georgia,serif;border-left:4px solid #C9A84C;padding-left:12px;">${emoji} ${title}</p>`;
  }

  return `
    <!-- INTRO -->
    <p style="margin:0 0 8px 0;font-size:13px;color:#C9A84C;text-transform:uppercase;letter-spacing:2px;font-family:Arial,sans-serif;text-align:center;">Austin Real Estate Market Update</p>
    <p style="margin:0 0 24px 0;font-size:22px;font-weight:700;color:#111827;font-family:Georgia,serif;text-align:center;">${data.month}</p>

    ${divider()}

    <!-- MARKET SNAPSHOT -->
    ${heading('📊', 'Market Snapshot')}

    <!-- Row 1 -->
    <table width="100%" cellpadding="0" cellspacing="0" border="0" style="border-collapse:separate;border-spacing:8px 0;margin-bottom:8px;">
      <tr>
        ${card('Median Home Price', data.medianPrice, data.medianPriceTrend)}
        <td width="3%" style="font-size:0;">&nbsp;</td>
        ${card('Homes Sold', data.homesSold, data.homesSoldTrend)}
        <td width="3%" style="font-size:0;">&nbsp;</td>
        ${card('Days on Market', data.daysOnMarket, data.daysOnMarketTrend)}
      </tr>
    </table>

    <!-- Row 2 -->
    <table width="100%" cellpadding="0" cellspacing="0" border="0" style="border-collapse:separate;border-spacing:8px 0;margin-bottom:28px;">
      <tr>
        ${card('Sold-to-List Ratio', data.soldListRatio, data.soldListRatioTrend)}
        <td width="3%" style="font-size:0;">&nbsp;</td>
        ${card('Price / Sq Ft', data.pricePerSqFt, data.pricePerSqFtTrend)}
        <td width="3%" style="font-size:0;">&nbsp;</td>
        ${card('Active Inventory', data.inventory, data.inventoryTrend)}
      </tr>
    </table>

    ${divider()}
