export function buildMarketUpdateNewsletterHtml(data: {
  month: string;
  medianPrice: string; medianPriceTrend: string;
  homesSold: string; homesSoldTrend: string;
  daysOnMarket: string; daysOnMarketTrend: string;
  soldListRatio: string; soldListRatioTrend: string;
  pricePerSqFt: string; pricePerSqFtTrend: string;
  inventory: string; inventoryTrend: string;
  marketSummary: string;
  neighborhoodSpotlight: string;
  buyerInsights: string;
  sellerInsights: string;
  tipOfMonth: string;
}): string {

  function stat(label: string, value: string, trend: string): string {
    return `<td style="width:30%;padding:12px;background:#F8F9FA;border:1px solid #E5E7EB;border-radius:6px;text-align:center;vertical-align:top;">
      <p style="margin:0 0 6px;font-size:10px;color:#6B7280;text-transform:uppercase;letter-spacing:1px;font-family:Arial,sans-serif;">${label}</p>
      <p style="margin:0 0 4px;font-size:22px;font-weight:700;color:#111827;font-family:Georgia,serif;">${value || 'N/A'}</p>
      <p style="margin:0;font-size:11px;color:#6B7280;font-family:Arial,sans-serif;">${trend || '&nbsp;'}</p>
    </td>`;
  }

  function bullets(text: string): string {
    if (!text) return '<li>See full report for details.</li>';
    return text.split('|').map(b => b.trim()).filter(Boolean)
      .map(b => `<li style="margin-bottom:8px;">${b}</li>`).join('');
  }

  return `
    <h2 style="font-family:Georgia,serif;font-size:13px;color:#C9A84C;text-transform:uppercase;letter-spacing:2px;margin:0 0 4px;text-align:center;">Austin Real Estate Market Update</h2>

    <table width="100%" cellpadding="0" cellspacing="8" border="0" style="margin-bottom:8px;">
      <tr>
        ${stat('Median Home Price', data.medianPrice, data.medianPriceTrend)}
        <td width="4%"></td>
        ${stat('Homes Sold', data.homesSold, data.homesSoldTrend)}
        <td width="4%"></td>
        ${stat('Days on Market', data.daysOnMarket, data.daysOnMarketTrend)}

    <table width="100%" cellpadding="0" cellspacing="8" border="0" style="margin-bottom:24px;">

    <h2 style="font-family:Georgia,serif;font-size:18px;color:#111827;border-left:4px solid #C9A84C;padding-left:10px;margin:0 0 12px;">🏘️ Neighborhood Spotlight</h2>
